// bgfx Android lifecycle (NativeActivity).
//
// Unlike the bgfx DESKTOP template — which owns a linear
// `pub fn main() { init; while (!shouldClose) { tick; render } }` body —
// the Android shell (`backends/bgfx/src/android_app.zig`) owns the event +
// frame loop. The NativeActivity glue calls our exported `android_main`
// on a dedicated app thread; from there `android_app.run(app)` pumps the
// ALooper, brings bgfx up on `INIT_WINDOW`, and calls back into the game
// once per frame (`tick_fn`) plus once at surface-ready (`init_fn`).
//
// So this template inverts the desktop body into three callbacks:
//   * `gameInit`  — engine + scene init, fired ONCE after bgfx is live
//                   (registered via `android_app.setInitCallback`).
//   * `gameFrame` — one engine frame: tick + render (registered via
//                   `android_app.setTickCallback`).
//   * `android_main` — the C entry the glue calls. Registers the two
//                   callbacks, then hands the loop to `android_app.run`.
//
// `labelle_provides_android_main` tells the backend's `android_app.zig` to
// suppress ITS own `android_main` export (otherwise a duplicate symbol);
// the game owns the entry so it can register callbacks before the loop.
const android_app = @import("backend_app");
const BackendGfxType = @import("backend_gfx");

pub const labelle_provides_android_main = true;

// Route zig `std.log` to Android logcat (tag "labelle"). Without a `logFn`,
// std.log writes to stderr — invisible in `adb logcat` — so engine/backend
// diagnostics never reach the device log (e.g. the surface_lost/surface_restored
// markers and "first surface (cold init)" from the bgfx-android lifecycle).
// bgfx's own C logging routes separately, which is why only bgfx lines showed
// before. This is set on the generated `main.zig` (the root module), so it
// covers ALL std.log across engine + backend + game. (#386 Phase 4 observability.)
pub const std_options: std.Options = .{
    .logFn = androidLogFn,
};

extern "c" fn __android_log_write(prio: c_int, tag: [*:0]const u8, text: [*:0]const u8) c_int;

fn androidLogFn(
    comptime level: std.log.Level,
    comptime scope: @EnumLiteral(),
    comptime format: []const u8,
    args: anytype,
) void {
    _ = scope;
    var buf: [1024]u8 = undefined;
    const text = std.fmt.bufPrintZ(&buf, format, args) catch {
        _ = __android_log_write(5, "labelle", "(log line truncated)");
        return;
    };
    // ANDROID_LOG_*: DEBUG=3, INFO=4, WARN=5, ERROR=6.
    const prio: c_int = switch (level) {
        .debug => 3,
        .info => 4,
        .warn => 5,
        .err => 6,
    };
    _ = __android_log_write(prio, "labelle", text);
}

// ── Device verification hooks (labelle-assembler#737) ────────────────────
// The desktop template resolves `LABELLE_FIXED_DT` and `LABELLE_SCREENSHOT_PATH`
// into loop-locals; the Android shell owns the loop, so the equivalents live at
// file scope and are resolved ONCE in `gameInit` — the frame path never pays
// for a lookup. Mirrors how `wasm.txt` hoists its `fixed_dt`.
//
// These are only reachable when the APK is built `debuggable` (the opt-in
// `.android = .{ .debuggable = true }` knob): the platform's `wrap.<package>`
// property is what gives the process a real environment, and it is honoured
// only for debuggable apps. On a release APK every one of these stays at its
// default and the whole block folds to today's behaviour — no screenshot, no
// fixed timestep, one extra never-taken branch per frame.
//
// Why it matters: without them, on-device checking falls back to `adb screencap`
// at wall-clock times, which on some devices applies a display colour transform
// (so colour cannot be judged) and has ~1s of latency (so a frame cannot be
// pinned to a simulation time).

/// Pinned simulation timestep, or `null` to measure the real frame period
/// (the default, and the only value a release build can have).
var fixed_dt: ?f32 = null;
/// Pending one-shot capture request from `LABELLE_SCREENSHOT_PATH`.
/// An ABSOLUTE path is taken verbatim, so the caller decides exactly where it
/// lands; a RELATIVE one is resolved against — and confined to — the app's own
/// files dir, so `adb pull` (or `run-as ... cat`) can fetch it afterwards. See
/// `resolveScreenshotPath`.
var screenshot_req: ?engine.ScreenshotRequest = null;
/// Under a fixed timestep the delay is translated into a FRAME INDEX up front,
/// so the captured frame is decided by simulated time, not by the clock — that
/// is what makes two runs byte-identical. `null` keeps the wall-clock gate.
var capture_frame: ?u64 = null;
/// 1-based count of simulated frames, feeding `capture_frame`.
var frame_index: u64 = 0;
/// Wall-clock baseline for the non-fixed-dt gate. Set at the END of `gameInit`,
/// once the project's setup and scene load are done — a baseline taken before
/// them would let slow device-side init eat the requested delay.
var loop_start_ns: i128 = 0;
/// Backing store for a RELATIVE `LABELLE_SCREENSHOT_PATH` resolved against the
/// activity's own internal data dir (see `resolveScreenshotPath`).
var screenshot_path_buf: [512]u8 = undefined;

/// ── Knob channel ─────────────────────────────────────────────────────────
/// `LABELLE_*` knobs, resolved PER KNOB: the process environment first (desktop
/// parity; reachable where the platform's `wrap.<package>` property is
/// permitted), and otherwise an optional `labelle_env` file in the activity's
/// OWN internal data dir. See `knobValue`.
///
/// The file exists because the environment channel is not always available: an
/// activity launched normally inherits zygote's environment, and `wrap.<package>`
/// — the one property that can replace it — is refused by SELinux on some
/// devices (measured on a Samsung SM-T505 / Android 12: `setprop` returns
/// "error code: 0x18") and caps its value at 92 bytes where it is allowed. The
/// app's private dir has neither problem, and writing to it requires `run-as`,
/// which the platform grants only for a `debuggable` APK. The RUNTIME read is
/// gated on the same opt-in independently, by asking the running apk for its
/// own `FLAG_DEBUGGABLE` (`android_app.isDebuggable()`) — `run-as` restricts
/// creation only, so a file left behind by a debuggable build must not keep
/// working after an update to a release build.
///
///     adb shell run-as <package> sh -c 'printf "LABELLE_FIXED_DT=0.02\n\
///       LABELLE_SCREENSHOT_PATH=shot\nLABELLE_SCREENSHOT_AFTER_SEC=2\n" \
///       > files/labelle_env'
///
/// Format: one `KEY=VALUE` per line; `#` starts a comment; blank lines ignored.
/// Missing file, unreadable file and malformed lines all degrade to "knob unset"
/// — a verification aid must never be able to break a run.
var knob_file_buf: [4096]u8 = undefined;
var knob_file: []const u8 = &.{};

fn loadKnobFile() void {
    // Gate on the RUNNING apk's own `android:debuggable` flag, not on the fact
    // that `run-as` was once able to WRITE this file. `run-as` restricts
    // creation only: install a debuggable build, drop a knob file, then update
    // to a release build over it, and the file survives in the app's private
    // dir — a release APK would go on honouring it, taking a fixed timestep
    // and writing screenshots. `android_app.isDebuggable()` asks the process
    // about its own `ApplicationInfo.FLAG_DEBUGGABLE` and fails closed, so a
    // release build ignores a stale file. `@hasDecl` so an older backend still
    // builds — there it degrades to "no file channel", never to "ungated".
    if (comptime !@hasDecl(android_app, "isDebuggable")) return;
    if (!android_app.isDebuggable()) return;
    const dir_c = android_app.internalDataPath() orelse return;
    var path_buf: [512]u8 = undefined;
    const path = std.fmt.bufPrintZ(&path_buf, "{s}/labelle_env", .{std.mem.span(dir_c)}) catch return;
    const fd = std.c.open(path.ptr, .{ .ACCMODE = .RDONLY }, @as(std.c.mode_t, 0));
    if (fd < 0) return;
    defer _ = std.c.close(fd);
    const n = std.c.read(fd, &knob_file_buf, knob_file_buf.len);
    if (n <= 0) return;
    knob_file = knob_file_buf[0..@intCast(n)];
    std.log.info("labelle: device knobs loaded from {s} ({d} bytes)", .{ path, knob_file.len });
}

/// The parse itself lives in the backend (`window.knobFromText`) so it is
/// unit-tested there rather than only exercised on a device.
fn knobFromFile(name: []const u8) ?[]const u8 {
    if (comptime !@hasDecl(window, "knobFromText")) return null;
    return window.knobFromText(knob_file, name);
}

/// Resolve `raw` into a path the process can actually write.
///
/// An ABSOLUTE path is taken verbatim, and deliberately so: it is unambiguous,
/// it names no directory implicitly, and it is the only way to aim a capture at
/// e.g. a debugger-writable external dir. Nothing is confined about it — the
/// caller spelled out exactly where the file goes.
///
/// A RELATIVE one is joined to `/data/data/<package>/files`, the only place a
/// NativeActivity can always write and the place `run-as <package> cat` can
/// read back. That matters because the ONLY channel for handing a debuggable
/// Android app an environment is the `wrap.<package>` system property, whose
/// value is capped at 92 bytes: spelling the absolute path in the knob leaves
/// no room for `LABELLE_FIXED_DT` beside it.
///
/// A relative path is CONFINED to that directory. Plain concatenation let
/// `../shared_prefs/state` resolve to a real, writable file outside the
/// capture area, so the screenshot would clobber unrelated app-private state.
/// `window.relativeKnobPathEscapes` walks the components and rejects anything
/// whose `..` reaches at or above the base dir (and anything resolving to the
/// dir itself, which names no file).
///
/// Returns null (and logs) when the name escapes, does not fit, or the activity
/// has no data path, so a bad knob degrades to "no screenshot", never a crash
/// and never a write somewhere else.
fn resolveScreenshotPath(raw: []const u8) ?[:0]const u8 {
    if (raw.len == 0) return null;
    if (raw[0] == '/') {
        return std.fmt.bufPrintZ(&screenshot_path_buf, "{s}", .{raw}) catch {
            std.log.err("LABELLE_SCREENSHOT_PATH too long: {s}", .{raw});
            return null;
        };
    }
    if (comptime @hasDecl(window, "relativeKnobPathEscapes")) {
        if (window.relativeKnobPathEscapes(raw)) {
            std.log.err("LABELLE_SCREENSHOT_PATH '{s}' escapes the app's private dir — ignored", .{raw});
            return null;
        }
    }
    const dir_c = android_app.internalDataPath() orelse {
        std.log.err("LABELLE_SCREENSHOT_PATH '{s}' is relative but the activity has no internal data path", .{raw});
        return null;
    };
    return std.fmt.bufPrintZ(&screenshot_path_buf, "{s}/{s}", .{ std.mem.span(dir_c), raw }) catch {
        std.log.err("LABELLE_SCREENSHOT_PATH too long once resolved: {s}", .{raw});
        return null;
    };
}

/// `LABELLE_FIXED_DT`'s accept/reject policy — the backend's own, so the file
/// channel and the environment channel cannot disagree about what a valid
/// timestep is.
fn parseKnobDt(raw: []const u8) ?f32 {
    if (comptime !@hasDecl(window, "parseFixedDtKnob")) return null;
    return window.parseFixedDtKnob(raw);
}

/// `LABELLE_SCREENSHOT_AFTER_SEC`'s accept/reject policy, the backend's own so
/// both channels agree. Notably rejects `inf`, which parses and is not
/// negative but makes `elapsed >= after_sec` false forever — the run would
/// capture nothing and merely look hung.
fn parseKnobAfterSec(raw: []const u8) ?f32 {
    if (comptime !@hasDecl(window, "parseAfterSecKnob")) return null;
    return window.parseAfterSecKnob(raw);
}

/// Resolve ONE knob: the process ENVIRONMENT first (desktop parity), then the
/// device knob file.
///
/// Per-knob on purpose. Resolving the whole screenshot request from whichever
/// channel happened to carry the PATH meant a path exported in the environment
/// plus a delay written to the knob file silently dropped the delay, and the
/// capture fired on frame 1. Each knob now picks its own channel.
fn knobValue(comptime name: [:0]const u8) ?[]const u8 {
    if (comptime @hasDecl(window, "knobFromEnv")) {
        if (window.knobFromEnv(name.ptr)) |v| return v;
    }
    return knobFromFile(name);
}

/// The project's DESIGN CANVAS, in design pixels — `.width`/`.height` from
/// `project.labelle`. `pub` so GAME CODE can read it (labelle-bgfx#42).
///
/// Content authored against a fixed canvas needs to know that canvas at
/// runtime, and until now nothing exposed it: the engine and gfx have no
/// accessor, and these consts were private to the generated root. A game
/// that hardcoded "1024" instead then broke the moment the canvas changed —
/// which is exactly how #42 was reported (a `screen_fill` backdrop sized
/// for 1024 leaving the rest of a wider canvas uncovered).
///
/// This is the DESIGN size, NOT the physical framebuffer. A `screen_fill`
/// layer stretches the canvas onto the whole surface, so content that
/// covers the canvas covers the surface at any window size; the physical
/// size is `window.width()`/`window.height()`.
pub const screen_w: u32 = {{width}};
pub const screen_h: u32 = {{height}};
const screen_title = "{{title}}";
const target_fps: u32 = {{fps}};

// {{platform_comment}}
// One-shot surface-ready init: bgfx is already up against the
// ANativeWindow surface (the shell called `window.initWindow` on
// INIT_WINDOW just before firing this). Build the game, wire hooks, and
// run the project's setup/scene-load code. Runs on the app thread, once.
fn gameInit() callconv(.c) void {
    const allocator = gpa.allocator();
    g = AssembledGame.init(allocator);

    g.setHooks(&hooks);
    // The engine works in the DESIGN canvas (config height) — the gfx
    // backend aspect-fits design→surface (setScreenSize/setDesignSize in
    // gameFrame), so the engine must NOT be told the raw surface height or
    // its coordinate math conflicts with the backend's fit. Matches the
    // sokol Android template (which uses the config `screen_h`, not the
    // ANativeWindow size). (An earlier revision fed the real surface height
    // here — correct only before the backend had a design-size system.)
    g.setScreenHeight(@as(f32, @floatFromInt(screen_h)));

    // Script Runtime Contract bind (labelle-engine#749 / labelle-assembler#593):
    // hand the game to the engine's script_contract once, right after Game.init
    // and BEFORE the assembler's init code below runs the scripting plugin's
    // setup (which boots the VM against the exported C surface). Double
    // @hasDecl gate — engine exposes the contract AND the assembler generated
    // the `scripting_enabled` flag — mirroring the desktop template (and the
    // fullscreen/vsync drain convention), so older engines and script-less
    // projects build unchanged. The per-frame twin,
    // `script_contract.drainEvents`, rides the assembler's tick_code
    // splice in gameFrame — between the plugin ticks and g.dispatchEvents().
    if (comptime @hasDecl(engine, "script_contract") and @hasDecl(@This(), "scripting_enabled")) {
        engine.script_contract.bind(&g);
        // Paired teardown BEFORE g.deinit(): the contract owns
        // subscription/inbox/emit-arena state through g.allocator and
        // must release it (and drop the game pointer) while the game
        // is still alive — otherwise the debug allocator reports the
        // contract's state as leaked and the exports stay bound to a
        // dying game.
    }
    defer if (comptime @hasDecl(engine, "script_contract") and @hasDecl(@This(), "scripting_enabled")) {
        engine.script_contract.unbind();
    };
    // Resolve the verification knobs once (see their declarations above).
    // Both are `@hasDecl`/env gated, so this is a no-op on a release APK and on
    // backends predating the helpers.
    loadKnobFile();
    if (comptime @hasDecl(window, "fixedDt")) fixed_dt = window.fixedDt();
    if (fixed_dt == null) {
        if (knobValue("LABELLE_FIXED_DT")) |raw| fixed_dt = parseKnobDt(raw);
    }
    // Path and delay resolve INDEPENDENTLY, each environment-first then file.
    // Resolving both from whichever channel carried the path meant an exported
    // `LABELLE_SCREENSHOT_PATH` plus a file-written delay dropped the delay and
    // fired the capture on frame 1.
    const shot_path: ?[]const u8 = knobValue("LABELLE_SCREENSHOT_PATH");
    // A rejected delay (unparseable, negative, NaN, or `inf`) falls back to
    // "capture as soon as possible" rather than to a gate that never fires.
    const shot_after: f32 = blk: {
        const raw = knobValue("LABELLE_SCREENSHOT_AFTER_SEC") orelse break :blk 0.0;
        break :blk parseKnobAfterSec(raw) orelse {
            std.log.err("LABELLE_SCREENSHOT_AFTER_SEC '{s}' is not a finite, non-negative number — capturing immediately", .{raw});
            break :blk 0.0;
        };
    };
    if (shot_path) |raw| {
        if (resolveScreenshotPath(raw)) |resolved| {
            screenshot_req = .{ .path = resolved, .after_sec = shot_after };
        }
    }
    if (comptime @hasDecl(window, "captureFrameFor")) {
        if (fixed_dt) |fdt| if (screenshot_req) |req| {
            capture_frame = window.captureFrameFor(req.after_sec, fdt);
        };
    }

{{preview_setup}}{{init_code}}
    // Start the screenshot delay clock HERE, after the project's setup and
    // scene load have run — not before them. Scene/plugin init on a device is
    // slow (asset upload, script VM boot), and a baseline taken ahead of it
    // spends the delay on work the capture was never meant to measure: with
    // `--after=2` and a 1.5 s init, the capture lands 0.5 s into the run. The
    // fixed-timestep path is unaffected either way (it counts SIMULATED frames
    // from the first tick, and never reads this clock), which is exactly why
    // the wall-clock path has to be anchored to the same instant: the first
    // frame.
    loop_start_ns = engine.nowNs();
}

// Per-frame tick: advance the engine, then render through bgfx. Mirrors
// the desktop template's loop body (tick → beginFrame → render →
// endFrame). The shell only calls this while the surface is live and
// the activity is resumed, so no `isRunning()`/close handling is needed
// here — the activity lifecycle drives shutdown.
fn gameFrame() callconv(.c) void {
    // Frame-rate-independent sim: derive dt from the real frame period
    // (`window.frameDuration` measures it with a monotonic clock) instead of
    // a fixed 0.016, so game speed doesn't drift if the device renders below
    // its target rate. Clamp to 4 frames at target_fps so a stall (app
    // backgrounded, GC pause) can't explode physics. Mirrors desktop.txt.
    // `fixed_dt` (LABELLE_FIXED_DT, labelle-bgfx#59 / labelle-assembler#737)
    // short-circuits the measurement when the caller has explicitly asked for a
    // reproducible run; it is null on every normal run and on every release
    // APK, so the measured branch below stays the default.
    const dt: f32 = fixed_dt orelse @min(@as(f32, @floatCast(window.frameDuration())), 4.0 / @as(f32, @floatFromInt(@max(target_fps, 1))));
    // Count the frame whose tick is about to run, so the capture gate below
    // (which sits AFTER `g.tick`) sees a 1-based count of SIMULATED frames.
    frame_index += 1;
    // Engine works in the DESIGN canvas (config height); the gfx backend
    // owns the design→surface aspect-fit below. Keep these consistent with
    // gameInit and the sokol Android template.
    g.setScreenHeight(@as(f32, @floatFromInt(screen_h)));
    // Feed the gfx backend the REAL surface size (physical) + the design
    // canvas (config), so it aspect-fits the 800x600-style design space
    // into the device's actual resolution. Per-frame because the surface
    // can change (rotation, pause/resume). Mirrors the sokol Android template.
    BackendGfxType.setScreenSize(window.width(), window.height());
    BackendGfxType.setDesignSize(@intCast(screen_w), @intCast(screen_h));
{{preview_heartbeat}}{{tick_code}}    g.tick(dt);

    window.beginFrame();
    window.clearBackground(245, 245, 245, 255);
    g.render();
    g.renderGizmos();
{{gui_draw_code}}
    // Screenshot capture (labelle-assembler#737). bgfx's `requestScreenShot` is
    // ASYNC and must be queued BEFORE the swap so it captures THIS frame, so the
    // call sits between the draw calls and `endFrame` — exactly where the desktop
    // template puts it. There is no surfaceless path on Android: the activity
    // always has a real ANativeWindow surface, so `captureHeadless` (offscreen
    // framebuffer readback) does not apply here.
    //
    // One-shot: `screenshot_req` is cleared on the capture frame. We deliberately
    // do NOT quit afterwards — an Android activity has no `--ticks` exit, and the
    // harness stops the app once the file has landed.
    if (comptime @hasDecl(window, "takeScreenshot")) {
        if (screenshot_req) |req| {
            const fire = if (capture_frame) |target|
                // Fixed-timestep run: fire on the frame the delay was translated
                // into, so the captured world state is identical every run. No
                // clock is consulted at all — this is what makes two captures at
                // the same fixed sim time byte-identical on device.
                frame_index >= target
            else blk: {
                const elapsed_sec: f32 = @as(f32, @floatFromInt(engine.nowNs() - loop_start_ns)) / 1_000_000_000.0;
                break :blk elapsed_sec >= req.after_sec;
            };
            if (fire) {
                window.takeScreenshot(req.path);
                screenshot_req = null;
            }
        }
    }
    window.endFrame();
}

// GPU-context-loss trampolines (epic #386 Phase 4). On Android,
// APP_CMD_TERM_WINDOW destroys the GPU surface + every bgfx texture/shader,
// and APP_CMD_INIT_WINDOW later recreates it. The bgfx shell fires these via
// `setSurfaceLostCallback` / `setSurfaceRestoredCallback`; we forward to the
// engine's `Game.surfaceLost()` / `surfaceRestored()`. `@hasDecl`-gated so an
// OLDER engine without those methods still compiles (the call folds away).
fn gameSurfaceLost() callconv(.c) void {
    if (@hasDecl(@TypeOf(g), "surfaceLost")) g.surfaceLost();
}
fn gameSurfaceRestored() callconv(.c) void {
    if (@hasDecl(@TypeOf(g), "surfaceRestored")) g.surfaceRestored();
}

// {{entry_comment}}
// The NativeActivity glue (`android_native_app_glue.c`, compiled into the
// `android_app` module) declares `extern void android_main(struct
// android_app*)` and calls it on the app thread. We register the game's
// init + frame callbacks, then hand the event/frame loop to the shell.
export fn android_main(app: *android_app.android_app) callconv(.c) void {
    android_app.setInitCallback(&gameInit);
    android_app.setTickCallback(&gameFrame);
    // GPU-context-loss callbacks (#386 Phase 4): the shell fires these around
    // the TERM_WINDOW teardown / INIT_WINDOW restore so the engine can drop
    // and rebuild its GPU catalog (sprites) across a surface cycle.
    android_app.setSurfaceLostCallback(&gameSurfaceLost);
    android_app.setSurfaceRestoredCallback(&gameSurfaceRestored);
{{android_backend_register}}{{immersive_register}}    android_app.run(app);
}
